<script>
document.addEventListener('DOMContentLoaded', function() {
    const variations = [
      '{% link /assets/2019/404/spinning.html %}',
      '{% link /assets/2019/404/typewriter.html %}',
      '{% link /assets/2019/404/ascii.html %}',
      '{% link /assets/2019/404/hello.html %}',
    ];
    
    /* Get previously selected variation to avoid repeats */
    const lastVariation = localStorage.getItem('last404Variation');
    
    /* Reorder array to put last variation first, then pick from remaining */
    if (lastVariation && variations.length > 1) {
        const lastIndex = variations.indexOf(lastVariation);
        if (lastIndex !== -1) {
            /* Move last variation to front */
            variations.splice(lastIndex, 1);
            variations.unshift(lastVariation);
        }
    }
    
    /* Pick randomly from positions 1 onwards (skipping index 0) */
    const availableCount = variations.length > 1 ? variations.length - 1 : variations.length;
    const randomIndex = Math.floor(Math.random() * availableCount) + (variations.length > 1 ? 1 : 0);
    const selectedVariation = variations[randomIndex];
    
    /* Console logging for debugging */
    console.log('404 Variation:', selectedVariation.split('/').pop());
    
    /* Store this selection for next time */
    localStorage.setItem('last404Variation', selectedVariation);
    const errorContainer = document.getElementById('error-content');
    
    /* Add loading state */
    errorContainer.classList.add('loading');
    
    /* Fetch and load the selected variation */
    fetch(selectedVariation)
        .then(response => {
            if (!response.ok) {
                throw new Error('Network response was not ok');
            }
            return response.text();
        })
        .then(html => {
            /* Remove loading state and update content */
            errorContainer.classList.remove('loading');
            errorContainer.innerHTML = html;
            
            /* Execute any scripts in the loaded content */
            const scripts = errorContainer.querySelectorAll('script');
            scripts.forEach(script => {
                const newScript = document.createElement('script');
                newScript.textContent = script.textContent;
                document.head.appendChild(newScript);
            });
            
            /* Log which variation was loaded (for debugging) */
            console.log('404 Variation loaded:', selectedVariation.split('/').pop());
        })
        .catch(error => {
            /* Remove loading state and keep fallback content */
            errorContainer.classList.remove('loading');
            console.log('Using fallback 404, error:', error);
            /* Fallback content already loaded */
        });
});
</script>
