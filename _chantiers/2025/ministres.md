<script>
        document.addEventListener('DOMContentLoaded', function() {

        const imageElements = document.querySelectorAll('.minister-card > img');
        let imagesProcessing = imageElements.length;
        
        /* Show loading cursor */
        document.body.classList.add('loading');
        
        function imageProcessed() {
            imagesProcessing--;
            if (imagesProcessing === 0) {
                /* All images processed, remove loading cursor */
                document.body.classList.remove('loading');
            }
        }

        imageElements.forEach(img => {
            const card = img.parentNode;
            const label = card.querySelector('p');
            if (label) {
                label.style.visibility = 'hidden';
            }

            const imgSrc = img.src;
            const container = document.createElement('div');
        
        container.style.display = 'block';
        if (img.parentNode) {
            img.parentNode.insertBefore(container, img);
            img.style.display = 'none';
        }

        const sketch = (p) => {
            let p5Image;
            p.preload = () => { p5Image = p.loadImage(imgSrc); };
            
            p.setup = () => {
                const displayWidth = 500;
                const aspectRatio = p5Image.height / p5Image.width;
                const displayHeight = Math.round(displayWidth * aspectRatio);
                
                p.createCanvas(displayWidth, displayHeight);
                
                const scaledImage = p.createImage(displayWidth, displayHeight);
                scaledImage.copy(p5Image, 0, 0, p5Image.width, p5Image.height, 0, 0, displayWidth, displayHeight);
                
                const ditheredImage = floydSteinbergDithering(scaledImage, p);
                p.image(ditheredImage, 0, 0);
                p.noLoop();
                if (label) {
                    label.style.visibility = 'visible';
                }
                /* Signal that this image is processed */
                imageProcessed();
            };
        };
        new p5(sketch, container);
    });

    function floydSteinbergDithering(imageToDither, p) {
        const dithered = imageToDither.get();
        dithered.filter(p.GRAY);
        dithered.filter(p.BLUR, 0.1);
        
        dithered.loadPixels();
        for (let i = 0; i < dithered.pixels.length; i += 4) {
            dithered.pixels[i] = Math.min(255, dithered.pixels[i] * 1.1);
            dithered.pixels[i + 1] = Math.min(255, dithered.pixels[i + 1] * 1.1);
            dithered.pixels[i + 2] = Math.min(255, dithered.pixels[i + 2] * 1.1);
        }
        dithered.updatePixels();
        

        const imageSize = dithered.width * dithered.height;
        const factor = imageSize < 100000 ? 12 : (imageSize < 200000 ? 8 : 6);
        
        for (let y = 0; y < dithered.height - 1; y++) {
            for (let x = 1; x < dithered.width - 1; x++) {
                const index = (x + y * dithered.width) * 4;
                const oldR = dithered.pixels[index];
                const newR = p.round(factor * oldR / 255) * (255 / factor);
                dithered.pixels[index] = newR;
                dithered.pixels[index + 1] = newR;
                dithered.pixels[index + 2] = newR;
                const errR = oldR - newR;
                addError(dithered, errR, x + 1, y, 7 / 16.0);
                addError(dithered, errR, x - 1, y + 1, 3 / 16.0);
                addError(dithered, errR, x, y + 1, 5 / 16.0);
                addError(dithered, errR, x + 1, y + 1, 1 / 16.0);
            }
        }
        dithered.updatePixels();
        return dithered;
    }

    function addError(dithered, err, x, y, quant) {
        if (x < 0 || x >= dithered.width || y < 0 || y >= dithered.height) return;
        const index = (x + y * dithered.width) * 4;
        dithered.pixels[index] += err * quant;
        dithered.pixels[index + 1] += err * quant;
        dithered.pixels[index + 2] += err * quant;
    }
});
